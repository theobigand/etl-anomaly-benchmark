import hashlib,platform
from pathlib import Path
import numpy as np,pandas as pd,torch
from etl_benchmark.quality.deep import VAEDetector

 d=VAEDetector(device='mps',epoch_num=1,batch_size=10,encoder_neuron_list=[4],decoder_neuron_list=[4],latent_dim=2).fit(frame);s=d.score(frame);print('OK',len(s),'finite',bool(np.isfinite(s).all()))
