import hashlib,platform
from pathlib import Path
import numpy as np,pandas as pd,torch
from etl_benchmark.quality.deep import DIFDetector
p=Path('src/etl_benchmark/quality/_torch_models.py');print('source',hashlib.sha256(p.read_bytes()).hexdigest(),'python',platform.python_version(),'platform',platform.platform(),'torch',torch.__version__,'mps_available',torch.backends.mps.is_available())
rng=np.random.default_rng(4);frame=pd.DataFrame(rng.normal(size=(30,3)),columns=['a','b','c'])

 d=DIFDetector(device='mps',n_ensemble=1,n_estimators=1,representation_dim=3,hidden_neurons=[4],batch_size=10).fit(frame);s=d.score(frame);print('OK',len(s),'finite',bool(np.isfinite(s).all()))
