import importlib.util
import sys
from pathlib import Path
import pytest

module_name, function, mutant, test = sys.argv[1:]
sys.path.insert(0, str(Path('research').resolve()))
spec = importlib.util.spec_from_file_location(module_name, f'mutants/research/{module_name}.py')
module = importlib.util.module_from_spec(spec)
sys.modules[module_name] = module
spec.loader.exec_module(module)
setattr(module, function, getattr(module, mutant))
raise SystemExit(pytest.main(['-q', test]))
