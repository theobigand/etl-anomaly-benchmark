import pandas as pd
from etl_benchmark.quality.window_drift import WindowDriftMonitor

def test_nullable_numeric_columns_can_be_scored():
    reference = pd.DataFrame({'time':pd.date_range('2024-01-01',periods=3,tz='UTC'), 'value':pd.Series([1,2,3],dtype='Int64'), 'measurement':pd.Series([1.,2.,3.],dtype='Float64')})
    future = pd.DataFrame({'time':pd.date_range('2024-01-04',periods=2,tz='UTC'), 'value':pd.Series([2,4],dtype='Int64'), 'measurement':pd.Series([2.,4.],dtype='Float64')})
    result = WindowDriftMonitor('time',['value','measurement'],window_size=2).fit(reference).evaluate(future)
    assert len(result)==1
    assert result.iloc[0]['n_rows']==2
